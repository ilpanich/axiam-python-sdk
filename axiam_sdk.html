axiam_sdk

Official Python client SDK for AXIAM (Access eXtended Identity and Authorization Management).

Re-exports the public surface: the sync AxiamClient and async AsyncAxiamClient REST entry points, the AuthError/AuthzError/ NetworkError exception taxonomy (CONTRACT.md §2), and the request/response models (LoginResult, User, AccessCheck, AccessResult, BatchCheckResult, UserInfo). See CONTRACT.md §1-§10 for the cross-language behavioral contract this SDK conforms to.

This module MUST remain importable with ONLY the runtime dependencies declared in [project.dependencies] (httpx, grpcio, aio-pika, pydantic, PyJWT) — the optional web-framework integrations (axiam_sdk.fastapi, axiam_sdk.django, see [project.optional-dependencies]) MUST NOT be imported from here.

477    def refresh(self) -> None:
478        """``POST /api/v1/auth/refresh`` (CONTRACT.md §1), routed through
479        the shared single-flight guard (§9) so concurrent 401s collapse
480        into exactly one in-flight refresh call. A 401 on the refresh call
481        itself is ``AuthError`` with no retry (§9.3, Pitfall 4)."""
482        observed_access = self._session.cookie_value(ACCESS_COOKIE)
483        if not observed_access:
484            raise AuthError("no access token to refresh — call login() first")

486        tenant_id, org_id = self._refresh_identifiers(observed_access)

560    def _authz_post_sync(self, path: str, body: dict[str, Any]) -> dict[str, Any]:
561        """POST an authz request body to *path*, transparently retrying once
562        via :meth:`_retry_after_refresh_sync` on a 401 (§9.3), and returning
563        the parsed JSON response body. Raises the mapped ``AxiamError``
564        family exception (CONTRACT.md §2) for any other non-2xx status."""
565        request = self._session.sync_client.build_request("POST", path, json=body)
566        response = self._session._send_sync(request)
567
568        if response.status_code == httpx.codes.UNAUTHORIZED:
569            response = self._retry_after_refresh_sync(request)
570
571        if response.status_code < 200 or response.status_code >= 300:
572            raise error_from_http_status(
573                response.status_code, "authz check failed", response=response
574            )
575        result: dict[str, Any] = response.json()
576        return result

578    def _retry_after_refresh_sync(self, original_request: httpx.Request) -> httpx.Response:
579        """On a 401, refresh exactly once (via the shared single-flight
580        guard) then retry the failed authz call exactly once. A second
581        failure propagates through the caller's own status check (§9.3, no
582        retry loop)."""
583        self.refresh()
584        retry_request = self._session.sync_client.build_request(
585            original_request.method,
586            original_request.url,
587            content=original_request.content,
588            headers={
589                k: v
590                for k, v in original_request.headers.items()
591                if k.lower() not in ("content-length", "x-csrf-token")
592            },
593        )
594        return self._session._send_sync(retry_request)

660    def oidc_refresh(
661        self,
662        *,
663        refresh_token: SecretStr | str,
664        scope: str | None = None,
665        tenant_id: str | None = None,
666        configuration: OidcConfiguration | None = None,
667    ) -> OidcTokenSet:
668        """``POST /oauth2/token`` with ``grant_type=refresh_token``
669        (CONTRACT.md §12.1) — refresh an ``OidcTokenSet``.
670
671        A **distinct operation** from :meth:`refresh`, which drives the
672        cookie/opaque-token session path at ``POST /api/v1/auth/refresh``
673        (§12.1 "``oidc_refresh`` vs ``refresh``") — the two are never
674        merged, aliased, or made to fall back to one another. Concurrent
675        ``oidc_refresh`` calls collapse into exactly one wire call, and the
676        selected caller's wire call additionally runs inside the same §9
677        single-flight guard the cookie-session :meth:`refresh` uses, so an
678        ``oidc_refresh`` and a concurrent cookie-session refresh can never
679        interleave.
680        """
681
682        def do_refresh() -> OidcTokenSet:
683            """Perform the actual ``refresh_token`` grant call; run inside
684            :meth:`~axiam_sdk.token.refresh_guard.RefreshGuard.run_exclusive_sync`
685            by ``under_guard``, and de-duplicated across concurrent callers
686            by the single-flight coalescer below."""
687            config = configuration or self.oidc_discover()
688            form = self._refresh_form(refresh_token=refresh_token, scope=scope)
689            url = self._token_endpoint_url(config, tenant_id)
690            request = self._session.sync_client.build_request("POST", url, data=form)
691            response = self._session._send_sync(request)
692            # No nonce: rule 6 does not apply to a refresh-issued ID token.
693            return self._handle_token_response(response, config, None)
694
695        def under_guard() -> OidcTokenSet:
696            """Run ``do_refresh`` inside the shared §9 refresh-guard lock,
697            so it can never interleave with a concurrent cookie-session
698            :meth:`refresh` call."""
699            token_set: OidcTokenSet = self._session.refresh_guard.run_exclusive_sync(do_refresh)
700            return token_set
701
702        result: OidcTokenSet = self._oidc_refresh_single_flight_sync.run(under_guard)
703        return result

156        for state in expired:
